```bash

git clone https://github.com/taoma888/awesome-ai-skills.git
cd awesome-ai-skills

# 2. 安装依赖
pip install requests

# 3. 变现机会挖掘
python3 skills/autonomous-revenue-machine/scripts/revenue_miner.py

# 4. 生成社媒内容
python3 skills/social-traffic-siphon/scripts/content_generator.py -t "Claude 4" -u "https://aisstt.fun"

# 5. 追踪收入
python3 skills/revenue-dashboard/scripts/revenue_tracker.py --add affiliate 500 pending

```
